\newcommand{\signatureImage}[3]{%
  \IfFileExists{#1}{\includegraphics[width=#2,height=#3,keepaspectratio]{#1}}{\fbox{\parbox[c][#3][c]{#2}{Firma pendiente}}}%
}

  \end{minipage}%
}
