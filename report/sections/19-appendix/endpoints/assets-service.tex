\multicolumn{3}{|l|}{\textit{Cosmetics}} \\
\hline
\texttt{GET}    & \texttt{/assets/cosmetics/categories}                          & Retorna la lista de todas las categorías de cosméticos. \\
\hline
\texttt{POST}   & \texttt{/assets/cosmetics/categories}                          & Crea una nueva categoría de cosmético. Solo administradores. \\
\hline
\texttt{GET}    & \texttt{/assets/cosmetics/categories/\{id\}}                   & Retorna la lista paginada de cosméticos de una categoría. Soporta filtros opcionales por \texttt{world\_id} y \texttt{player\_id}. \\
\hline
\texttt{PUT}    & \texttt{/assets/cosmetics/categories/\{id\}}                   & Sube un nuevo sprite a una categoría como cosmético. Requiere ownership del world o ser administrador. \\
\hline
\texttt{PUT}    & \texttt{/assets/cosmetics/categories/\{id\}/} \newline \texttt{sprites/\{sprite\_id\}} & Registra un cosmético en una categoría reutilizando el URL de un sprite existente. Requiere ownership del world o ser administrador. \\
\hline
\texttt{GET}    & \texttt{/assets/cosmetics/\{id\}}                              & Retorna un cosmético por su ID. \\
\hline
\texttt{DELETE} & \texttt{/assets/cosmetics/\{id\}}                              & Elimina un cosmético por su ID. Requiere ownership. \\
\hline
\texttt{GET}    & \texttt{/assets/cosmetics/worlds/\{world\_id\}}                & Retorna la lista paginada de cosméticos asociados a un world. \\
\hline
\texttt{GET}    & \texttt{/assets/cosmetics/economy-summary}                     & Retorna conteos agregados y precio promedio de cosméticos. \\
\hline
\texttt{GET}    & \texttt{/assets/internal/cosmetics/} \newline \texttt{\{cosmetic\_id\}} & Retorna datos internos de un cosmético (precio, creador). Uso interno entre servicios. \\
\hline
\texttt{POST}   & \texttt{/assets/internal/users/\{user\_id\}} \newline \texttt{/cosmetics} & Registra la compra de un cosmético para un usuario. Uso interno entre servicios. \\
\hline
\multicolumn{3}{|l|}{\textit{Items}} \\
\hline
\texttt{GET}    & \texttt{/assets/items/\{id\}}                                  & Retorna un item por su ID. \\
\hline
\texttt{DELETE} & \texttt{/assets/items/\{id\}}                                  & Elimina un item por su ID. Requiere ownership. \\
\hline
\texttt{GET}    & \texttt{/assets/items/world/\{world\_id\}}                     & Retorna la lista de todos los items de un world. \\
\hline
\texttt{PUT}    & \texttt{/assets/items/world/\{world\_id\}}                     & Sube uno o más sprites de items asociados a un world (multipart). \\
\hline
\multicolumn{3}{|l|}{\textit{Materials}} \\
\hline
\texttt{GET}    & \texttt{/assets/materials}                                     & Retorna la lista paginada de materials. Si se provee \texttt{world\_id}, incluye los materials del world más los \textit{default}; si no, solo los \textit{default}. \\
\hline
\texttt{GET}    & \texttt{/assets/materials/\{id\}}                              & Retorna un material por su ID. \\
\hline
\texttt{DELETE} & \texttt{/assets/materials/\{id\}}                              & Elimina un material por su ID. Requiere ownership. \\
\hline
\texttt{PUT}    & \texttt{/assets/materials/world/\{world\_id\}}                 & Sube uno o más materials con nombre y archivo asociados a un world (multipart). Solo administradores para materials \textit{default}. \\
\hline
\multicolumn{3}{|l|}{\textit{Models}} \\
\hline
\texttt{GET}    & \texttt{/assets/models/world/\{world\_id\}}                    & Retorna la lista de modelos 3D de un world más los modelos \textit{default}. \\
\hline
\texttt{PUT}    & \texttt{/assets/models/world/\{world\_id\}}                    & Sube un modelo 3D (\texttt{.glb}) asociado a un world. Solo administradores para modelos \textit{default}. \\
\hline
